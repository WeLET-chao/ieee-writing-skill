% Problem formulation: spec vector, pass/fail metrics.

\begin{algorithm}[t]
    \caption{Automated spec-to-schematic flow}
    \label{alg:flow}
    \begin{algorithmic}[1]
        \REQUIRE Specification $\mathbf{s}$, template $\mathcal{T}$, simulator, optimizer
        \ENSURE Architecture $x$, sizing $\mathbf{w}$, status
        \STATE $x \leftarrow \textsc{Generate}(\mathcal{T}, \mathbf{s})$
        \STATE Validate $x$ with behavioral simulation; regenerate until pass or budget exhausted
        \STATE $\mathbf{w} \leftarrow \textsc{Optimize}(x, \mathbf{s})$
        \RETURN $(x, \mathbf{w}, \textsc{SignOffCheck}(x, \mathbf{w}, \mathbf{s}))$
    \end{algorithmic}
\end{algorithm}
